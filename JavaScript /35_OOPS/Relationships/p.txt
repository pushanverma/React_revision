```js
// ============================================================
// UNIVERSITY MANAGEMENT SYSTEM
// ============================================================
//
// OOP Concepts Covered:
//
// 1. Classes & Objects
// 2. Constructors
// 3. Instance Methods
// 4. Getters & Setters
// 5. Encapsulation
// 6. Private Fields (#)
// 7. Inheritance
// 8. extends
// 9. super()
// 10. Method Overriding
// 11. Polymorphism
// 12. Static Properties & Methods
// 13. IS-A Relationship
// 14. HAS-A Relationship
// 15. Association
// 16. Dependency
// 17. Aggregation
// 18. Composition
// 19. Dependency Injection
// 20. instanceof
// 21. Prototype Chain
//

// ============================================================
// 1. PERSON
// ============================================================
//
// Person is our BASE / PARENT class.
//
// Student IS-A Person
// Professor IS-A Person
//
// This demonstrates:
// - Class
// - Constructor
// - Encapsulation
// - Private field
// - Getter
// - Setter
// - Instance method
// ============================================================

class Person {

  // Private property
  #age;

  constructor(name, age) {
    this.name = name;
    this.#age = age;
  }

  // -------------------------
  // Getter
  // -------------------------

  get age() {
    return this.#age;
  }

  // -------------------------
  // Setter
  // -------------------------

  set age(value) {

    if (value < 0) {
      console.log("Age cannot be negative.");
      return;
    }

    this.#age = value;
  }

  // -------------------------
  // Normal method
  // -------------------------

  introduce() {
    return `Hello, my name is ${this.name}`;
  }
}



// ============================================================
// 2. ADDRESS
// ============================================================
//
// Address is an independent class.
//
// Student HAS-A Address
//
// This demonstrates:
// - HAS-A relationship
// - Object reference
// ============================================================

class Address {

  constructor(city, state) {
    this.city = city;
    this.state = state;
  }

  getAddress() {
    return `${this.city}, ${this.state}`;
  }
}



// ============================================================
// 3. COURSE
// ============================================================
//
// Course contains students and a professor.
//
// Course ASSOCIATES with:
// - Student
// - Professor
//
// ============================================================

class Course {

  constructor(courseName, courseCode) {

    this.courseName = courseName;
    this.courseCode = courseCode;

    // List of students enrolled
    this.students = [];

    // Professor who teaches the course
    this.professor = null;
  }

  // -------------------------
  // Add Student
  // -------------------------

  addStudent(student) {

    this.students.push(student);

    console.log(
      `${student.name} has been added to ${this.courseName}`
    );
  }

  // -------------------------
  // Assign Professor
  // -------------------------

  setProfessor(professor) {

    this.professor = professor;

    console.log(
      `${professor.name} is now teaching ${this.courseName}`
    );
  }

  // -------------------------
  // Course Details
  // -------------------------

  getCourseDetails() {

    return `${this.courseCode} - ${this.courseName}`;
  }

  // -------------------------
  // Display Students
  // -------------------------

  showStudents() {

    console.log(`\nStudents in ${this.courseName}:`);

    this.students.forEach((student) => {

      console.log(
        `- ${student.name} (${student.studentId})`
      );

    });
  }
}



// ============================================================
// 4. NOTIFICATION SERVICE
// ============================================================
//
// This class will be used as a DEPENDENCY.
//
// Student does not own NotificationService.
//
// Student simply USES it when needed.
//
// Relationship:
//
// Student
//    |
//    | temporarily uses
//    ↓
// NotificationService
//
// ============================================================

class NotificationService {

  send(message) {

    console.log(`\nNotification: ${message}`);
  }
}



// ============================================================
// 5. STUDENT
// ============================================================
//
// Student extends Person.
//
// Therefore:
//
// Student IS-A Person
//
// This demonstrates:
// - Inheritance
// - extends
// - super()
// - Method overriding
// - Polymorphism
// - HAS-A relationship
// - Association
// - Dependency Injection
// ============================================================

class Student extends Person {

  constructor(
    name,
    age,
    studentId,
    address,
    notificationService
  ) {

    // Call Person constructor
    super(name, age);

    this.studentId = studentId;

    // -------------------------
    // HAS-A relationship
    // -------------------------
    //
    // Student HAS-A Address
    //

    this.address = address;


    // -------------------------
    // Dependency Injection
    // -------------------------
    //
    // NotificationService is injected
    // into Student.
    //

    this.notificationService =
      notificationService;


    // -------------------------
    // Student HAS-A Courses
    // -------------------------

    this.courses = [];
  }


  // ==========================================================
  // Method Overriding
  // ==========================================================
  //
  // Person has introduce()
  //
  // Student provides its own version.
  //
  // This is POLYMORPHISM.
  // ==========================================================

  introduce() {

    return (
      super.introduce() +
      ` and I am a student with ID ${this.studentId}`
    );
  }


  // ==========================================================
  // Enroll in Course 
  // ==========================================================
  //
  // Student ASSOCIATES with Course.
  //
  // ==========================================================

  enroll(course) {

// Adding the course to Students list 
    this.courses.push(course);

// Adding the Student to the Course 
    course.addStudent(this);

    console.log(
      `${this.name} enrolled in ${course.courseName}`
    );
  }


  // ==========================================================
  // Show Student Courses
  // ==========================================================

  showCourses() {

    console.log(`\nCourses for ${this.name}:`);

    this.courses.forEach((course) => {

      console.log(
        `- ${course.getCourseDetails()}`
      );

    });
  }


  // ==========================================================
  // Dependency Example
  // ==========================================================
  //
  // Student uses NotificationService.
  //
  // ==========================================================

  notify() {

    this.notificationService.send(
      `${this.name}, your university notification has arrived.`
    );
  }
}



// ============================================================
// 6. PROFESSOR
// ============================================================
//
// Professor IS-A Person.
//
// This demonstrates:
// - Inheritance
// - Method overriding
// - Association with Course
// ============================================================

class Professor extends Person {

  constructor(name, age, employeeId) {

    super(name, age);

    this.employeeId = employeeId;
  }


  // ==========================================================
  // Method Overriding
  // ==========================================================

  introduce() {

    return (
      super.introduce() +
      ` and I am a Professor with employee ID ${this.employeeId}`
    );
  }


  // ==========================================================
  // Teach Course
  // ==========================================================
  //
  // Professor ASSOCIATES with Course.
  //
  // ==========================================================

  teach(course) {

    course.setProfessor(this);
  }
}



// ============================================================
// 7. DEPARTMENT
// ============================================================
//
// Department contains Professors and Courses.
//
// Department AGGREGATES Professors.
//
// Why?
//
// Professors were created independently:
//
// const professor1 = new Professor(...);
//
// Then we put them into Department.
//
// Therefore Professor can exist independently.
//
// ============================================================

class Department {

  constructor(name, professors = []) {

    this.name = name;

    // Aggregation
    this.professors = professors;

    // Department HAS-A Courses
    this.courses = [];
  }


  // ==========================================================
  // Add Professor
  // ==========================================================

  addProfessor(professor) {

    this.professors.push(professor);

    console.log(
      `${professor.name} joined ${this.name} department`
    );
  }


  // ==========================================================
  // Add Course
  // ==========================================================

  addCourse(course) {

    this.courses.push(course);

    console.log(
      `${course.courseName} added to ${this.name} department`
    );
  }


  // ==========================================================
  // Show Professors
  // ==========================================================

  showProfessors() {

    console.log(
      `\nProfessors in ${this.name}:`
    );

    this.professors.forEach((professor) => {

      console.log(
        `- ${professor.name}`
      );

    });
  }
}



// ============================================================
// 8. UNIVERSITY 
// ============================================================
//
// University contains Departments.
//
// This demonstrates:
// - Composition
// - Static members
// - HAS-A relationship
//
// ============================================================

class University {

  // ==========================================================
  // Static Property
  // ==========================================================
  //
  // Belongs to the CLASS, not individual objects.
  //
  // ==========================================================

  static universityCount = 0;


  constructor(name) {

    this.name = name;

    // HAS-A relationship
    this.departments = [];

    // Increase number of universities
    University.universityCount++;
  }


  // ==========================================================
  // Static Method 
  // ==========================================================

  static getUniversityCount() {

    return University.universityCount;
  }


  // ==========================================================
  // Add Department
  // ==========================================================
  //
  // University HAS-A Department.
  //
  // ==========================================================

  addDepartment(department) {

    this.departments.push(department);

    console.log(
      `${department.name} department added to ${this.name}`
    );
  }


  // ==========================================================
  // Show Departments
  // ==========================================================

  showDepartments() {

    console.log(
      `\nDepartments in ${this.name}:`
    );

    this.departments.forEach((department) => {

      console.log(
        `- ${department.name}`
      );

    });
  }
}
```
